```gdscript
const SEEDLING_MAX_LEVEL := 3

static func _race_awakened(state: GameState, race_id: StringName) -> bool:
	return state.races.has(race_id) and bool(state.races[race_id].get("awakened", false))

static func buy_seedling(state: GameState) -> bool:
	if state.seedling_level >= SEEDLING_MAX_LEVEL:
		return false
	var cost := BigNum.new(float(CostCalculator.seedling_cost(state.seedling_level)))
	if not state.sap.is_greater_or_equal(cost):
		return false
	state.sap.sub(cost)
	state.seedling_level += 1
	return true

static func _buy_facility(state: GameState, race_id: StringName, level_field: String, cost_func: Callable) -> bool:
	if not _race_awakened(state, race_id):
		return false
	var level := int(state.get(level_field))
	var cost := BigNum.new(float(cost_func.call(level)))
	if not state.sap.is_greater_or_equal(cost):
		return false
	state.sap.sub(cost)
	state.set(level_field, level + 1)
	return true

static func buy_firepit(state: GameState) -> bool:
	return _buy_facility(state, &"human", "firepit_level", CostCalculator.firepit_cost)

static func buy_ring(state: GameState) -> bool:
	return _buy_facility(state, &"forestfolk", "ring_level", CostCalculator.ring_cost)

static func buy_forge(state: GameState) -> bool:
	return _buy_facility(state, &"stoneborn", "forge_level", CostCalculator.forge_cost)

static func buy_totem_pole(state: GameState) -> bool:
	return _buy_facility(state, &"wildfolk", "totem_pole_level", CostCalculator.totem_pole_cost)

static func buy_deep_dream(state: GameState) -> bool:
	if state.deep_dream:
		return false
	var cost := BigNum.new(3000.0)
	if not state.sap.is_greater_or_equal(cost):
		return false
	state.sap.sub(cost)
	state.memory.add(BigNum.new(15.0))
	state.deep_dream = true
	return true

static func buy_wind_veil(state: GameState) -> bool:
	if state.wind_veil:
		return false
	var cost := BigNum.new(2500.0)
	if not state.sap.is_greater_or_equal(cost):
		return false
	state.sap.sub(cost)
	state.faith.add(BigNum.new(30.0))
	state.wind_veil = true
	return true
```

var _checks := 0
var _fails := 0

func _check(name: String, cond: bool) -> void:
	_checks += 1
	if cond:
		print("PASS  ", name)
	else:
		_fails += 1
		print("FAIL  ", name)

func _init() -> void:
	var gm = load("res://autoloads/game_manager.gd").new()
	gm._state = GameState.new()
	var s := gm.get_state()
	# 1) 嫩叶教学链 3 级封顶
	s.sap = BigNum.new(1000.0)
	_check("嫩叶 L1", gm.buy_seedling())
	_check("嫩叶 L2", gm.buy_seedling())
	_check("嫩叶 L3", gm.buy_seedling())
	_check("嫩叶封顶", not gm.buy_seedling())
	_check("嫩叶 3 级", s.seedling_level == 3)
	# 2) 一次性大额 + 叙事
	s.sap = BigNum.new(6000.0)
	_check("深根梦", gm.buy_deep_dream())
	_check("深根梦记忆+15", s.memory.to_value() == 15.0)
	_check("深根梦幂等", not gm.buy_deep_dream())
	_check("风语膜", gm.buy_wind_veil())
	_check("风语膜信仰+30", s.faith.to_value() == 30.0)
	# 3) 四族设施（各族唤醒门）
	s.sap = BigNum.new(100000.0)
	s.races["human"] = {"awakened": true, "population": 50.0}
	_check("火塘需人族醒已醒", gm.buy_firepit())
	_check("林地未醒不可买歌之环", not gm.buy_ring())
	s.races["forestfolk"] = {"awakened": true, "population": 30.0}
	s.races["stoneborn"] = {"awakened": true, "population": 20.0}
	s.races["wildfolk"] = {"awakened": true, "population": 80.0}
	_check("歌之环", gm.buy_ring())
	_check("铸根坊", gm.buy_forge())
	_check("图腾柱", gm.buy_totem_pole())
	# 4) 设施 tick 产出（人族 pop50 会产记忆/信仰——差值或关 sap 增长）
	s.sap = BigNum.new(0.0)  # 冻结增长但设施照产
	var m0 := s.memory.to_value()
	var f0 := s.faith.to_value()
	var p0 := s.sap.to_value()
	gm._process(1.0)
	_check("火塘+图腾柱产记忆", s.memory.to_value() - m0 >= 0.2)  # 0.1+0.1（人族梦产另计）
	_check("歌之环产信仰", s.faith.to_value() - f0 >= 0.3)
	_check("铸根坊产树液", s.sap.to_value() - p0 >= 0.5)
	# 5) 存档往返 + 旧档
	var d := s.to_dict()
	var back := GameState.from_dict(d)
	_check("存档往返设施", back.firepit_level == s.firepit_level and back.seedling_level == 3)
	var old := GameState.from_dict({"tick": 5})
	_check("旧档回退", old.seedling_level == 0 and old.deep_dream == false)
	gm.free()
	print("E2E 结果: %d 检查, %d 失败" % [_checks, _fails])
	quit(1 if _fails > 0 else 0)
