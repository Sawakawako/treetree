# 里程碑 5f 灵魂系统 实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 落地灵魂系统——河底守恒模型（100 恒）+ 复活（1 灵魂 + 500 growth → 人口 +10）+ 夺魂（夺梦揭示后可抽，人口 -3 + 关系 -2），为明选引擎与终局牺牲之选提供三级资源基础。

**Architecture:** 纯逻辑 `SoulActions`（RefCounted 静态，与 PlunderActions/RelationActions 同模式）+ `GameState.soul_river` 序列化扩展 + GameManager 入口与信号（UI 只监听）+ UI 按钮（visible 动态控制，同夺梦/亲密模式）。守恒由实现保证：`revive` 河底 -1、`plunder_soul` 河底 +1（cap 100），测试验证 river 变化正确。

**Tech Stack:** Godot 4.7.1 mono / GDScript / GdUnit4 6.2.1（headless CLI）。

**Spec:** `docs/superpowers/specs/2026-08-31-world-tree-m5f-soul-design.md`（权威）；主线 `docs/superpowers/specs/2026-08-31-world-tree-design.md` §13.8（灵魂系统）+ §14.5（灵魂约 100 缕）。

## Global Constraints

- 资源一律 BigNum（禁裸 float 存资源）；平衡系数除外——但本里程碑 soul_river 是 **int**（设计 §3.1 明示，守恒计数非资源量），growth 用 BigNum（.sub）
- 逻辑类 RefCounted 纯函数可 headless 测；UI 只通过信号更新，不直改数据（Layer Cake 规则）
- 中文文件一律用 edit/write 工具或 .NET 显式 UTF-8 读写（禁 PowerShell 默认编码）
- 新增 class_name 脚本后必须先跑 `godot --headless --path . --import`（类注册）

```gdscript
class_name SoulActions
extends RefCounted

const RIVER_TOTAL := 100
const REVIVE_COST_SOUL := 1
const REVIVE_COST_GROWTH := 500.0
const REVIVE_POP_GAIN := 10
const PLUNDER_SOUL_POP_LOSS := 3
const PLUNDER_SOUL_RELATION_LOSS := 2

static func _race_awakened(state: GameState, race_id: StringName) -> bool:
	return state.races.has(race_id) and bool(state.races[race_id].get("awakened", false))

static func river(state: GameState) -> int:
	return state.soul_river

static func can_revive(state: GameState, race_id: StringName) -> bool:
	if not _race_awakened(state, race_id):
		return false
	if state.soul_river < REVIVE_COST_SOUL:
		return false
	return state.growth.is_greater_or_equal(BigNum.new(REVIVE_COST_GROWTH))

static func revive(state: GameState, race_id: StringName) -> Dictionary:
	if not can_revive(state, race_id):
		return {"ok": false}
	state.soul_river -= REVIVE_COST_SOUL
	state.growth.sub(BigNum.new(REVIVE_COST_GROWTH))
	var pop := float(state.races[race_id]["population"]) + float(REVIVE_POP_GAIN)
	state.races[race_id]["population"] = pop
	return {"ok": true, "race_id": race_id, "pop": pop}

static func can_plunder_soul(state: GameState, race_id: StringName) -> bool:
	if not _race_awakened(state, race_id):
		return false
	if PlunderActions.reveal_stage(state, race_id) < 1:
		return false
	var pop := float(state.races[race_id].get("population", 0.0))
	if pop < float(PLUNDER_SOUL_POP_LOSS):
		return false
	return state.soul_river < RIVER_TOTAL

static func plunder_soul(state: GameState, race_id: StringName) -> Dictionary:
	if not can_plunder_soul(state, race_id):
		return {"ok": false}
	state.soul_river = mini(state.soul_river + 1, RIVER_TOTAL)
	var pop := float(state.races[race_id]["population"]) - float(PLUNDER_SOUL_POP_LOSS)
	state.races[race_id]["population"] = pop
	var rel := RelationActions.apply_change(state, race_id, -PLUNDER_SOUL_RELATION_LOSS)
	return {"ok": true, "race_id": race_id, "pop": pop, "relation": rel}
```

```gdscript
extends SceneTree

var _checks := 0
var _fails := 0

func _check(name: String, cond: bool) -> void:
	_checks += 1
	if cond:
		print("PASS  ", name)
	else:
		_fails += 1
		print("FAIL  ", name, "  <-- 检查失败")

func _init() -> void:
	# 1) 唤醒人族 + 攒 growth
	var gm := load("res://autoloads/game_manager.gd").new()
	gm._state = GameState.new()
	gm._state.sap = BigNum.new(200.0)
	gm.explore_relic()
	gm._state.sap = BigNum.new(200.0)
	gm.explore_relic()
	gm._process(1.0)
	_check("人族唤醒", gm.is_human_awakened())
	# 2) 复活：growth 500 → 人口 +10，河底 99
	gm._state.growth = BigNum.new(500.0)
	var r := gm.revive_race(&"human")
	_check("复活成功", bool(r.get("ok", false)))
	_check("复活后河底 99", gm.get_state().soul_river == 99)
	_check("复活后人口 60", float(gm.get_state().races["human"]["population"]) == 60.0)
	# 3) 夺梦揭示（3 次）→ 夺魂可行
	for i in 3:
		gm.plunder_race(&"human")
	_check("夺梦揭示 1 级", PlunderActions.reveal_stage(gm.get_state(), &"human") == 1)
	var ps := gm.plunder_soul_race(&"human")
	_check("夺魂成功", bool(ps.get("ok", false)))
	_check("夺魂后河底回 100", gm.get_state().soul_river == 100)
	_check("夺魂后人口 57", float(gm.get_state().races["human"]["population"]) == 57.0)
	_check("夺魂后关系 -3", RelationActions.get_relation(gm.get_state(), &"human") == -3 if RelationActions.get_relation(gm.get_state(), &"human") <= -3 else RelationActions.get_relation(gm.get_state(), &"human") == -3)
	# 4) 存档往返
	var d := gm.get_state().to_dict()
	var back := GameState.from_dict(d)
	_check("存档往返 soul_river", back.soul_river == gm.get_state().soul_river)
	# 5) 旧档回退
	var old := GameState.from_dict({"tick": 5})
	_check("旧档回退 100", old.soul_river == 100)
	gm.free()
	print("E2E 结果: %d 检查, %d 失败" % [_checks, _fails])
	if _fails > 0:
		quit(1)
	else:
		quit(0)
```

> ⚠️ 第 4 项检查的夺魂关系断言写法笔误风险：`s.relations` 初始 0 → 夺魂 -2 → -2。脚本里那一行写复杂了，简化为 `_check("夺魂后关系 -2", RelationActions.get_relation(gm.get_state(), &"human") == -2)`。
