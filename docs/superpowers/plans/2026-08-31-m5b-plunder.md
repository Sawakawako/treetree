```gdscript
class_name PlunderActions
extends RefCounted

const REVEAL_THRESHOLDS := [3, 6, 9]

static func _race_awakened(state: GameState, race_id: StringName) -> bool:
	return state.races.has(race_id) and bool(state.races[race_id].get("awakened", false))

static func count(state: GameState, race_id: StringName) -> int:
	return int(state.plundered.get(race_id, 0))

static func reveal_stage(state: GameState, race_id: StringName) -> int:
	var n := count(state, race_id)
	if n >= REVEAL_THRESHOLDS[2]:
		return 3
	if n >= REVEAL_THRESHOLDS[1]:
		return 2
	if n >= REVEAL_THRESHOLDS[0]:
		return 1
	return 0

static func is_frozen(state: GameState, race_id: StringName) -> bool:
	return reveal_stage(state, race_id) >= 1

static func can_plunder(state: GameState, race_id: StringName) -> bool:
	return _race_awakened(state, race_id)

static func plunder(state: GameState, race_id: StringName) -> Dictionary:
	if not can_plunder(state, race_id):
		return {"ok": false}
	var data := PlunderData.get_data(race_id)
	var yield_mem := float(data.get("yield", 0.0))
	var before := reveal_stage(state, race_id)
	# 石裔无梦：不涨 counter、无产出
	if yield_mem > 0.0:
		state.memory.add(BigNum.new(yield_mem))
		state.plundered[race_id] = count(state, race_id) + 1
	var after := reveal_stage(state, race_id)
	var revealed := after > before
	if revealed:
		state.plunder_reveals.append(race_id)
		RelationActions.apply_change(state, race_id, -1)
		if race_id == &"wildfolk" and after == 1:
			# 惊扰：1 级揭示时人口 -20%（一次性，由 plunder_reveals 保证不重复）
			var pop := float(state.races[race_id].get("population", 0.0))
			state.races[race_id]["population"] = roundi(pop * 0.8)
		var reveals: Array = data.get("reveals", [])
		var idx := mini(after - 1, reveals.size() - 1)
		return {"ok": true, "memory": yield_mem, "text": str(reveals[maxi(idx, 0)]), "revealed": true}
	var text := PlunderData.signal_text(race_id, count(state, race_id))
	return {"ok": true, "memory": yield_mem, "text": text, "revealed": false}
```

extends SceneTree

func _init() -> void:
	var failures: Array[String] = []
	var s := GameState.new()
	s.races["human"] = {"awakened": true, "population": 50.0}
	s.races["wildfolk"] = {"awakened": true, "population": 100.0}

	for i in 9:
		PlunderActions.plunder(s, &"human")

		failures.append("memory not +9")
	if PlunderActions.reveal_stage(s, &"human") != 3:
		failures.append("stage not 3")
	if int(s.relations["human"]) != -1:
		failures.append("relation not -3 total")
	if s.plunder_reveals.size() != 1:
		failures.append("reveals not once")
	# 2. 冻结：reveal_stage >= 1 → tick 不增长
	var before := float(s.races["human"]["population"])
	s.sap = BigNum.new(1000.0)
	RaceManager.tick_races(s)
	if float(s.races["human"]["population"]) != before:
		failures.append("frozen race grew")
	# 3. 野民惊扰：3 次 → 人口 -20%（100 → 80）
	var s2 := GameState.new()
	s2.races["wildfolk"] = {"awakened": true, "population": 100.0}
	for i in 3:
		PlunderActions.plunder(s2, &"wildfolk")
	if float(s2.races["wildfolk"]["population"]) != 80.0:
		failures.append("wildfolk -20% failed")
	# 4. 石裔无梦：产出 0、counter 0
	var s3 := GameState.new()
	s3.races["stoneborn"] = {"awakened": true, "population": 20.0}

	if float(r.get("memory", -1.0)) != 0.0 or PlunderActions.count(s3, &"stoneborn") != 0:
		failures.append("stoneborn plunder failed")
	# 5. 存档往返 + 旧档回退
	var back := GameState.from_dict(s.to_dict())
	if int(back.plundered.get("human", 0)) != 9 or back.plunder_reveals.size() != 1:
		failures.append("save roundtrip")
	var legacy := GameState.from_dict({"races": {"human": {"awakened": true, "population": 50.0}}})
	if not legacy.plundered.is_empty():
		failures.append("legacy fallback")
	if failures.is_empty():
